posts.length === 0 ? (
              <EmptyState
                title="O silêncio reina"
                description="Esta alma ainda não compartilhou seus pensamentos na comunidade."
                icon={<MessageCircle size={64} className="text-primary/50 animate-float" />}
                action={isOwn ? (
                  <Button variant="primary" size="sm" onClick={() => document.querySelector('textarea')?.focus()}>
                    <MessageCircle size={14} className="mr-2" />
                    Escrever o primeiro post
                  </Button>
                ) : undefined}
                className="py-16 animate-fade-in-up"
              />
            ) : (
              <Stack gap="md" className="space-y-4">
{posts.map((post, index) => {
                    const delay = index * 80;
                    return (
                      <div
                        key={post.id}
                        className="animate-reveal-up"
                        style={{ '--animation-delay': delay + 'ms' } as React.CSSProperties }
                      >
                        <div className="p-4 bg-surface/50 backdrop-blur-sm border border-border rounded-xl">
                          <p className="text-text">{post.content}</p>
                        </div>
                      </div>
                    );
                  })}
              </Stack>
            </Container>
          </Section>
      </div>
    );
  }
}
